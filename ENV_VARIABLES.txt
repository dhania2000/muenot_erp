# Secret used to sign/verify login session cookies (JWT). Use a long random
# string, e.g. output of: openssl rand -base64 48

# longer be decrypted.

# 3. EMAIL - GENERIC FALLBACK  [OPTIONAL]
# =============================================================================
# These are used as a DEFAULT for any department that does NOT have its own
# department-specific variables set (section 4). If every department has its
# own values, you do not strictly need these - but it is good to keep one
# working fallback.
#
# Two ways to send mail (pick one per department):
#   A) SMTP        - standard mail server (ports 465/587)
#   B) Gmail API   - use this if your host BLOCKS outbound SMTP

# ---- A) Generic SMTP ----

SMTP_SECURE=true

SMTP_PASS=your-mailbox-password

# 7. SOCIAL MEDIA PUBLISHING (Marketing > Social)  [OPTIONAL]
# =============================================================================
# Needed only to connect real social accounts and publish to them. Each
# platform uses its OWN developer app with its OWN credentials - they are NOT
# interchangeable. Only the platforms you configure appear as connectable.
#
# The OAuth redirect URI for each platform is:
#   <APP_URL>/api/marketing/social/callback/<platform>
# e.g. https://erp.muenot.co.in/api/marketing/social/callback/linkedin
# Register that EXACT URL in each provider's developer console. Stored access
# and refresh tokens are encrypted at rest with SETTINGS_ENCRYPTION_KEY.
#
# Optional override if the redirect origin differs from APP_URL:
# SOCIAL_REDIRECT_BASE=https://erp.muenot.co.in

# ---- LinkedIn (Sign In with LinkedIn using OpenID Connect + Share on LinkedIn) ----
# Base scopes (always requested): openid profile email w_member_social.
#   -> Enables PERSONAL profile posting out of the box.
LINKEDIN_CLIENT_ID=your-linkedin-client-id
LINKEDIN_CLIENT_SECRET=your-linkedin-client-secret
#
# ---- LinkedIn COMPANY PAGE posting (optional, requires external approval) ----
# Posting as the Muenot Company Page needs the "Community Management API"
# product on your LinkedIn app. That is an approval in the LinkedIn Developer
# portal (developer.linkedin.com) — it CANNOT be enabled from code. Once your
# app is granted it, turn it on here so the extra scopes
# (r_organization_admin, w_organization_social) start being requested:
# LINKEDIN_ORG_ENABLED=true
#
# Optional: pin the exact org (the number in urn:li:organization:<id>).
# If unset, the app picks the admin org whose name/vanity matches
# LINKEDIN_ORGANIZATION_NAME (defaults to "muenot"), else the first admin org.
# LINKEDIN_ORGANIZATION_ID=
# LINKEDIN_ORGANIZATION_NAME=Muenot
#
# Advanced: fully override the requested scope list (space/comma separated).
# Including w_organization_social also implicitly enables company posting.
# LINKEDIN_SCOPES=openid profile email w_member_social r_organization_admin w_organization_social

# ---- X / Twitter (OAuth 2.0 with PKCE) ----
X_CLIENT_ID=your-x-client-id
X_CLIENT_SECRET=your-x-client-secret

# ---- Facebook (Page publishing via Graph API v21) ----
# Publishes as a Facebook Page the connecting user administers. Flow:
#   user authorizes -> list managed Pages (/me/accounts) -> pick a Page ->
#   use that Page's access token -> publish as the Page.
FACEBOOK_APP_ID=your-facebook-app-id
FACEBOOK_APP_SECRET=your-facebook-app-secret
#
# There are TWO kinds of Meta apps and they authorize DIFFERENTLY:
#
#   A) Classic Facebook Login app (Consumer/Business, standard "Facebook Login"
#      product). It accepts raw scopes. The app requests exactly:
#          pages_show_list  pages_read_engagement  pages_manage_posts
#      Add these under App > Use cases / Permissions and you are done. Do NOT
#      set FACEBOOK_CONFIG_ID for this kind of app.
#
#   B) "Facebook Login for Business" app. It REJECTS raw scopes with
#      "Invalid Scopes". Instead you create a Login configuration in the Meta
#      dashboard (which itself contains pages_show_list / pages_read_engagement
#      / pages_manage_posts) and pass its configuration id here. When set, the
#      app sends config_id and NO scope param:
# FACEBOOK_CONFIG_ID=your-facebook-login-configuration-id
#
# Advanced: override the exact scope list for the classic-login path
# (space/comma separated). Ignored when FACEBOOK_CONFIG_ID is set.
# FACEBOOK_SCOPES=pages_show_list pages_read_engagement pages_manage_posts

# ---- Instagram (Instagram API with Instagram Login) ----
# A DEDICATED Instagram app - do NOT reuse the Facebook app id/secret here,
# that causes PLATFORM_INVALID_APP_ID. Requires an Instagram Business or
# Creator account. Register the same callback URL (…/callback/instagram) as an
# Instagram "OAuth redirect URI".
INSTAGRAM_APP_ID=your-instagram-app-id
INSTAGRAM_APP_SECRET=your-instagram-app-secret


# =============================================================================
# 8. WHATSAPP BUSINESS (Marketing > WhatsApp)  [OPTIONAL]
# =============================================================================
# Needed only to connect a WhatsApp Business (Cloud API) number and use the
# WhatsApp inbox. The WABA ID, Phone Number ID and permanent access token are
# entered in the app UI (Marketing > WhatsApp > Integrate); the token is stored
# ENCRYPTED at rest with SETTINGS_ENCRYPTION_KEY (section 1), so make sure that
# key is set. The variables below power INBOUND delivery (the webhook) and are
# read from the server environment.

# Token Meta echoes back during the webhook verification handshake. Choose any
# hard-to-guess string and paste the SAME value into the Meta app dashboard
# under WhatsApp > Configuration > Webhook > "Verify token".
WHATSAPP_WEBHOOK_VERIFY_TOKEN=change-me-to-a-long-random-string

# App secret used to verify the signature (X-Hub-Signature-256) on inbound
# webhook POSTs. Without a valid secret, inbound messages are rejected. If it
# is unset, the app falls back to META_APP_SECRET, then FACEBOOK_APP_SECRET —
# so if you already set FACEBOOK_APP_SECRET (section 7) for the SAME Meta app,
# you can omit this.
WHATSAPP_APP_SECRET=your-meta-app-secret

# Optional: pin the Graph API version used for all WhatsApp calls. Defaults to
# a currently-supported version; only set this to override.
# WHATSAPP_GRAPH_VERSION=v23.0
